БЛОК 2. Настройки и запрос списка товаров
-----------------------------------------------------------------
DB = dict(host="localhost", port=3306, user="root", password="root", database="chudo_obuv", charset="utf8mb4")
COLUMNS = ("Наименование", "Категория", "Подкатегория", "Производитель", "Цена, руб.", "В наличии, пар")
PRODUCTS_SQL = """
SELECT p.name, c.name, s.name, m.name, p.price, COALESCE(SUM(st.quantity), 0)
FROM products p
JOIN categories c ON c.id = p.category_id
JOIN subcategories s ON s.id = p.subcategory_id
JOIN manufacturers m ON m.id = p.manufacturer_id
LEFT JOIN stock st ON st.product_id = p.id
GROUP BY p.id
ORDER BY p.id
"""
ORDER_COLUMNS = ("Номер заказа", "Дата", "Клиент", "Товар", "Размер", "Количество", "Цена, руб.")
ORDERS_SQL = """
SELECT o.id, o.order_date, CONCAT_WS(' ', u.last_name, u.first_name, u.patronymic), p.name, s.size, i.quantity, i.price
FROM orders o
JOIN users u ON u.id = o.user_id
JOIN order_items i ON i.order_id = o.id
JOIN products p ON p.id = i.product_id
JOIN sizes s ON s.id = i.size_id
ORDER BY o.id, i.id
"""

Предназначение: все настройки в одном месте. DB — параметры подключения
(используются в функции query), COLUMNS — заголовки таблицы товаров,
PRODUCTS_SQL — запрос, который собирает данные товара из нескольких таблиц
(так как в 3НФ категории, производители и т.д. хранятся отдельно).

Построчно:
  DB = dict(...)              — словарь параметров: сервер, порт, пользователь, пароль, имя базы,
                                кодировка utf8mb4 (чтобы русский текст читался правильно).
  COLUMNS = (...)             — кортеж названий столбцов таблицы в окне (по-русски).
  SELECT p.name, c.name, ...  — выбираем название товара, категорию, подкатегорию, производителя, цену
                                и сумму остатков по всем размерам; COALESCE(..., 0) — если остатков нет, будет 0.
  FROM products p             — основная таблица товаров, p — короткое имя (псевдоним).
  JOIN categories c ON ...    — подставляем название категории по category_id.
  JOIN subcategories s ON ... — подставляем название подкатегории по subcategory_id.
  JOIN manufacturers m ON ... — подставляем название производителя по manufacturer_id.
  LEFT JOIN stock st ON ...   — присоединяем остатки; LEFT — чтобы товар без остатков тоже показался.
  GROUP BY p.id               — группируем строки остатков по товару, чтобы SUM посчитал общее количество.
  ORDER BY p.id               — выводим в порядке id (как в исходном Excel).
  ORDER_COLUMNS = (...)       — заголовки таблицы заказов.
  SELECT o.id, o.order_date, CONCAT_WS(...), ... — номер и дата заказа, ФИО клиента (склеено через пробел),
                                название товара, размер, количество и цена из строки заказа.
  FROM orders o               — таблица заказов (шапка: номер, дата, клиент).
  JOIN users u ON ...         — подставляем ФИО клиента по user_id.
  JOIN order_items i ON ...   — присоединяем строки заказа (товары в заказе).
  JOIN products p ON ...      — подставляем название товара по product_id.
  JOIN sizes s ON ...         — подставляем размер по size_id.
  ORDER BY o.id, i.id         — сортировка по номеру заказа, внутри — по порядку строк.

БЛОК 3. Функция query
-----------------------------------------------------------------
def query(sql, args=()):
    with pymysql.connect(**DB) as connection, connection.cursor() as cursor:
        cursor.execute(sql, args)
        return cursor.fetchall()

Предназначение: единая функция для любого запроса к базе. Используется при
авторизации (метод login) и при загрузке таблиц товаров и заказов (метод table).

Построчно:
  def query(sql, args=()):         — принимает текст SQL и параметры (по умолчанию пустой кортеж).
  with pymysql.connect(**DB) ...   — открывает подключение с настройками DB (** распаковывает словарь)
                                     и курсор для выполнения запросов; with сам закроет их после блока.
  cursor.execute(sql, args)        — выполняет запрос; значения из args подставляются вместо %s
                                     безопасно (защита от SQL-инъекций).
  return cursor.fetchall()         — возвращает все строки результата в виде кортежей.

class App(tk.Tk):
    def __init__(self):
        super().__init__()
        self.geometry("1100x600")
        self.logo = tk.PhotoImage(file="logo.png").subsample(8)
        self.iconphoto(True, self.logo)
        self.show_login()

Предназначение: App — это главное окно приложения. При создании окна
задаются размер, логотип/иконка и сразу открывается экран входа.

Построчно:
  class App(tk.Tk):                 — класс наследуется от tk.Tk, то есть сам является окном.
  def __init__(self):               — конструктор, выполняется при создании App().

  self.geometry("1100x600")         — размер окна 1100 на 600 пикселей.
  self.logo = tk.PhotoImage(...)    — загружает logo.png; subsample(8) уменьшает картинку в 8 раз.
                                      Сохраняется в self, чтобы картинка не удалилась из памяти.
  self.iconphoto(True, self.logo)   — ставит логотип иконкой окна.
  self.show_login()                 — показывает экран авторизации.

    def show_login(self):

БЛОК 7. Метод login (авторизация)
-----------------------------------------------------------------
    def login(self, login, password):
        try:
            users = query("SELECT CONCAT_WS(' ', u.last_name, u.first_name, u.patronymic), r.name "
                          "FROM users u JOIN roles r ON r.id = u.role_id WHERE u.login = %s AND u.password = %s",
                          (login, password))
        except pymysql.Error as error:
            return messagebox.showerror("Ошибка подключения", f"Не удалось подключиться к базе данных:\n{error}")
        if not users:
            return messagebox.showerror("Ошибка входа", "Неверный логин или пароль")
        self.show_products(*users[0])

Предназначение: проверяет логин и пароль по таблице users. Если пользователь
найден — открывает список товаров с его ФИО и ролью; если нет — показывает ошибку.

Построчно:
  try:                               — начало блока, где может возникнуть ошибка подключения.
  users = query("SELECT CONCAT_WS(...) — CONCAT_WS склеивает фамилию, имя и отчество через пробел;
                                       r.name — название роли из таблицы roles;
                                       WHERE ... = %s — ищем строку с введённым логином и паролем.
  (login, password)                  — значения, которые подставятся вместо двух %s.
  except pymysql.Error as error:     — если MySQL не запущен или неверные настройки DB...
  return messagebox.showerror(...)   — ...показываем окно «Ошибка подключения» и выходим из метода.
  if not users:                      — запрос ничего не нашёл = логин или пароль неверны.
  return messagebox.showerror(...)   — окно «Ошибка входа».
  self.show_products(*users[0])      — users[0] = (ФИО, роль); * раскладывает кортеж на два аргумента.

БЛОК 10. Метод table (вывод таблицы из базы)
-----------------------------------------------------------------
    def table(self, columns, sql):
        table = ttk.Treeview(self, columns=columns, show="headings")
        for column in columns:
            table.heading(column, text=column)
        if "Наименование" in columns:
            table.column("Наименование", width=350)
        for row in query(sql):
            table.insert("", "end", values=row)
        table.pack(fill="both", expand=True, padx=10, pady=10)

Предназначение: общий метод, который строит таблицу по списку столбцов и
SQL-запросу. Используется и для товаров, и для заказов, чтобы не повторять код.

Построчно:
  table = ttk.Treeview(self, columns=columns, show="headings") — таблица с переданными столбцами;
                                      show="headings" скрывает лишний служебный первый столбец.
  for column in columns:            — для каждого столбца...
  table.heading(column, text=column)— ...задаём текст заголовка.
  if "Наименование" in columns:     — только в таблице товаров есть столбец «Наименование»...
  table.column("Наименование", width=350) — ...делаем его шире, чтобы длинные названия помещались.
  for row in query(sql):            — выполняем запрос и перебираем строки результата.
  table.insert("", "end", values=row) — добавляем строку в конец таблицы ("" — без родителя).
  table.pack(fill="both", expand=True, ...) — таблица занимает всё оставшееся место в окне.

ЧАСТЬ 2. БД_скрипт.txt (SQL)

-----------------------------------------------------------------
БЛОК 1. Создание базы
-----------------------------------------------------------------
DROP DATABASE IF EXISTS chudo_obuv;
CREATE DATABASE chudo_obuv CHARACTER SET utf8mb4;
USE chudo_obuv;

Предназначение: каждый запуск скрипта создаёт базу «с нуля», поэтому скрипт
можно выполнять повторно.

Построчно:
  DROP DATABASE IF EXISTS ...  — удаляет старую базу, если она есть.
  CREATE DATABASE ... utf8mb4  — создаёт новую базу с кодировкой, поддерживающей русский язык.
  USE chudo_obuv;              — все следующие команды выполняются в этой базе.


-----------------------------------------------------------------
БЛОК 2. Справочники
-----------------------------------------------------------------
CREATE TABLE roles (id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(50) NOT NULL UNIQUE);
CREATE TABLE categories (...);  CREATE TABLE subcategories (...);
CREATE TABLE manufacturers (...);  CREATE TABLE sizes (id ..., size DECIMAL(3,1) NOT NULL UNIQUE);

Предназначение: таблицы для повторяющихся значений (вынесены по 3НФ).
Основные таблицы ссылаются на них по id.

Построчно (на примере roles):
  id INT AUTO_INCREMENT PRIMARY KEY — первичный ключ, номер проставляется автоматически (1, 2, 3...).
  name VARCHAR(50) NOT NULL UNIQUE  — название, обязательно, без повторов.
  size DECIMAL(3,1)                 — размер обуви как число с одним знаком после запятой (36.5).


-----------------------------------------------------------------
БЛОК 3. Таблица users
-----------------------------------------------------------------
Предназначение: пользователи для авторизации в приложении.

Построчно:
  last_name, first_name, patronymic — фамилия, имя, отчество (разделены по 1НФ).
  login VARCHAR(50) NOT NULL UNIQUE — логин, не может повторяться.
  password ... DEFAULT '123456'     — пароль; если при вставке не указан, будет 123456.
  role_id INT NOT NULL              — номер роли.
  FOREIGN KEY (role_id) REFERENCES roles(id) — внешний ключ: роль должна существовать в roles.


-----------------------------------------------------------------
БЛОК 4. Таблица products
-----------------------------------------------------------------
Предназначение: товары. Текстовые повторяющиеся поля заменены на внешние ключи.

Построчно:
  name                         — наименование товара.
  category_id, subcategory_id, manufacturer_id — ссылки на справочники (FOREIGN KEY ниже).
  image VARCHAR(100)           — имя файла картинки (IMG_....png).
  description, composition TEXT— описание и состав (длинный текст).
  price DECIMAL(10,2)          — цена с копейками.


-----------------------------------------------------------------
БЛОК 5. Таблица stock
-----------------------------------------------------------------
Предназначение: остатки — сколько пар конкретного товара конкретного размера
доступно для заказа. Связь «многие ко многим» между products и sizes.

Построчно:
  product_id, size_id          — ссылки на товар и размер.
  quantity INT NOT NULL DEFAULT 0 — количество.
  UNIQUE (product_id, size_id) — одна пара товар+размер может быть записана только один раз.


-----------------------------------------------------------------
БЛОК 6. Таблицы orders и order_items
-----------------------------------------------------------------
Предназначение: заказы. Шапка заказа (номер, дата, клиент) хранится один раз
в orders, а товары заказа — отдельными строками в order_items.

Построчно:
  orders.order_date DATE        — дата заказа.
  orders.user_id                — клиент, ссылка на users(id).
  order_items.order_id          — к какому заказу относится строка, ссылка на orders(id).
  order_items.product_id, size_id — какой товар и какого размера заказан.
  order_items.quantity          — количество пар.
  order_items.price DECIMAL(10,2) — цена за единицу на момент заказа.


-----------------------------------------------------------------
БЛОК 7. INSERT INTO ...
-----------------------------------------------------------------
Предназначение: перенос данных из Excel в базу. Порядок важен: сначала
справочники, потом таблицы, которые на них ссылаются.

Построчно:
  INSERT INTO roles (name) VALUES ('Администратор'), ... — добавляет строки; id проставятся
                                                           автоматически по порядку (1, 2, 3).
  INSERT INTO users (..., role_id) VALUES (..., 1)       — числа в конце — id из справочника roles.
  INSERT INTO orders (id, ...)  — номер заказа указан явно, как в Excel, на него ссылается order_items.
  В products, stock и order_items аналогично стоят id категорий, подкатегорий, производителей, товаров и размеров.
